local function kartById(id: string): Kart
	return KART_BY_ID[id] or KARTS[1]
end

local function currencyName(currency: Currency, amount: number): string
	if currency == "credits" then
		return if amount == 1 then "Credit" else "Credits"
	end
	return if amount == 1 then "Gem" else "Gems"
end

local function priceText(part: Part): string
	return `{format.grouped(part.price)} {currencyName(part.currency, part.price)}`
end

local function balanceOf(snapshot: ServerState, currency: Currency): number
	return if currency == "credits" then snapshot.credits else snapshot.gems
end

local function isExpensive(part: Part): boolean
	if part.currency == "credits" then
		return part.price >= EXPENSIVE_CREDITS
	end
	return part.price >= EXPENSIVE_GEMS
end

local function ordinal(n: number): string
	local suffix = "th"
	local hundred = n % 100
	if hundred < 11 or hundred > 13 then
		local ten = n % 10
		if ten == 1 then
			suffix = "st"
		elseif ten == 2 then
			suffix = "nd"
		elseif ten == 3 then
			suffix = "rd"
		end
	end
	return `{n}{suffix}`
end

local function trim(text: string): string
	local result = string.gsub(text, "^%s+", "")
	result = string.gsub(result, "%s+$", "")
	return result
end

local function nameBefore(a: Kart, b: Kart): boolean
	local left, right = string.lower(a.name), string.lower(b.name)
	if left ~= right then
		return left < right
	end
	return a.id < b.id
end

local function filterKarts(query: string, kind: string, rarity: string, sort: string): { Kart }
	local needle = string.lower(trim(query))
	local list: { Kart } = {}
	for _, kart in KARTS do
		local kindOk = kind == "all" or kart.class == kind
		local rarityOk = rarity == "all" or kart.rarity == rarity
		local nameOk = needle == "" or string.find(string.lower(kart.name), needle, 1, true) ~= nil
		if kindOk and rarityOk and nameOk then
			table.insert(list, kart)
		end
	end
	table.sort(list, function(a: Kart, b: Kart): boolean
		if sort == "speed" and a.speed ~= b.speed then
			return a.speed > b.speed
		end
		if sort == "rarity" and a.rarity ~= b.rarity then
			return RARITY_RANK[a.rarity] > RARITY_RANK[b.rarity]
		end
		return nameBefore(a, b)
	end)
	return list
end

local function toast(message: string, key: string)
	if not alive then
		return
	end
	app.presentToast(function(ui: Facet.Controls): Instance
		return ui.Text "Message" { text = message, textRole = "strong" }
	end, { key = key, position = "top", width = "hug" })
end

-- Runs `work` as the server's answer unless the script was disposed or the
-- model was reset while the request was in flight.
local function afterServer(seconds: number, work: () -> ())
	local requestEpoch = epoch
	task.delay(seconds, function()
		if alive and requestEpoch == epoch then
			work()
		end
	end)
end

local function finishRace(session: number)
	local place = raceRandom:NextInteger(1, 5)
	local xpGain = 240 - place * 30
	local creditGain = 450 - place * 50
	local nextState = table.clone(state:peek())
	nextState.races += 1
	nextState.credits += creditGain
	nextState.xp += xpGain
	local leveled = false
	while nextState.xp >= nextState.xpGoal do
		nextState.xp -= nextState.xpGoal
		nextState.level += 1
		nextState.xpGoal += 250
		leveled = true
	end
	state:set(nextState)
	queue:set({ phase = "idle", since = 0, session = session + 1 })
	local kart = kartById(nextState.kart)
	lastRace:set(`Last race: {ordinal(place)} of 8 · +{xpGain} XP · +{format.grouped(creditGain)} Credits`)
	toast(`{kart.name} finished {ordinal(place)}! +{xpGain} XP, +{format.grouped(creditGain)} Credits`, "race")
	if leveled then
		toast(`Level up! You reached level {nextState.level}.`, "level")
	end
end

local function pressRace()
	local current = queue:peek()
	if current.phase == "idle" then
		local session = current.session + 1
		queue:set({ phase = "joining", since = os.clock(), session = session })
		afterServer(0.6, function()
			if queue:peek().session ~= session then
				return
			end
			queue:set({ phase = "queued", since = os.clock(), session = session })
			afterServer(7, function()
				local latest = queue:peek()
				if latest.session ~= session or latest.phase ~= "queued" then
					return
				end
				queue:set({ phase = "found", since = os.clock(), session = session })
				toast("Race found: Neon Circuit, 8 racers", "race")
				afterServer(3, function()
					if queue:peek().session ~= session then
						return
					end
					queue:set({ phase = "racing", since = os.clock(), session = session })
					afterServer(6, function()
						if queue:peek().session == session then
							finishRace(session)
						end
					end)
				end)
			end)
		end)
	elseif current.phase == "queued" then
		queue:set({ phase = "leaving", since = current.since, session = current.session })
		afterServer(0.4, function()
			if queue:peek().session ~= current.session then
				return
			end
			queue:set({ phase = "idle", since = 0, session = current.session + 1 })
			toast("You left the race queue.", "race")
		end)
	end
end

local function claimDaily()
	if dailyBusy:peek() then
		return
	end
	dailyBusy:set(true)
	dailyRefusal:set("")
	afterServer(0.8, function()
		dailyBusy:set(false)
		local snapshot = state:peek()
		if not snapshot.dailyReady then
			dailyRefusal:set("Already claimed today. Nothing changed; your next reward unlocks tomorrow.")
			return
		end
		local nextState = table.clone(snapshot)
		nextState.credits += DAILY_CREDITS
		nextState.gems += DAILY_GEMS
		nextState.dailyReady = false
		nextState.streak += 1
		nextState.nextDailyAt = os.time() + 20 * 3600
		state:set(nextState)
		dailyOpen:set(false)
		toast(`Daily reward claimed: +{format.grouped(DAILY_CREDITS)} Credits, +{DAILY_GEMS} Gems`, "daily")
	end)
end

local function chooseKart(kart: Kart)
	if kartPending:peek() ~= nil or state:peek().kart == kart.id then
		return
	end
	kartPending:set(kart.id)
	kartRefusal:set("")
	afterServer(0.6, function()
		kartPending:set(nil)
		local phase = queue:peek().phase
		if phase == "found" or phase == "racing" then
			kartRefusal:set("Karts lock once a race is found. Your kart is unchanged; try again after this race.")
			return
		end
		local nextState = table.clone(state:peek())
		nextState.kart = kart.id
		state:set(nextState)
		toast(`{kart.name} is set for your next race`, "kart")
	end)
end

local function sendPurchase(part: Part, done: (Reply) -> ())
	local pending = table.clone(purchasePending:peek())
	pending[part.id] = true
	purchasePending:set(pending)
	afterServer(0.8, function()
		local settled = table.clone(purchasePending:peek())
		settled[part.id] = nil
		purchasePending:set(settled)
		local snapshot = state:peek()
		if snapshot.owned[part.id] then
			done({ accepted = false, message = `You already own {part.name}. Nothing was spent.` })
			return
		end
		local balance = balanceOf(snapshot, part.currency)
		if balance < part.price then
			local missing = part.price - balance
			done({
				accepted = false,
				message = `Not enough {currencyName(part.currency, 2)}: you need {format.grouped(missing)} more. Nothing was spent.`,
			})
			return
		end
		local owned = table.clone(snapshot.owned)
		owned[part.id] = true
		local nextState = table.clone(snapshot)
		nextState.owned = owned
		if part.currency == "credits" then
			nextState.credits -= part.price
		else
			nextState.gems -= part.price
		end
		state:set(nextState)
		done({ accepted = true, message = `{part.name} added to your parts bay` })
	end)
end

local function requestBuy(part: Part)
	local snapshot = state:peek()
	if snapshot.owned[part.id] or purchasePending:peek()[part.id] then
		return
	end
	shopRefusal:set({ id = "", message = "" })
	if isExpensive(part) and balanceOf(snapshot, part.currency) >= part.price then
		confirmBusy:set(false)
		confirmRefusal:set("")
		confirmId:set(part.id)
		return
	end
	sendPurchase(part, function(reply: Reply)
		if reply.accepted then
			toast(reply.message, "purchase")
		else
			shopRefusal:set({ id = part.id, message = reply.message })
		end
	end)
end

local function confirmPurchase()
	local id = confirmId:peek()
	if id == nil or confirmBusy:peek() then
		return
	end
	local part = PART_BY_ID[id]
	confirmBusy:set(true)
	confirmRefusal:set("")
	sendPurchase(part, function(reply: Reply)
		confirmBusy:set(false)
		if reply.accepted then
			confirmId:set(nil)
			toast(reply.message, "purchase")
		else
			confirmRefusal:set(reply.message)
		end
	end)
end

local function sendChat(phrase: string)
	chatOpen:set(false)
	afterServer(0.35, function()
		local now = os.clock()
		if now - lastChatAt < 2.5 then
			toast("Slow down! Wait a moment before sending another message.", "chat")
			return
		end
		lastChatAt = now
		toast(`You: {phrase}`, "chat")
	end)
end

local function placed(points: { Facet.PathPoint }, cx: number, cy: number, scale: number): { Facet.PathPoint }
	local out: { Facet.PathPoint } = {}
	for _, p in points do
		table.insert(out, {
			x = cx + (p.x - 0.5) * scale,
			y = cy + (p.y - 0.5) * scale,
			inX = p.inX * scale,
			inY = p.inY * scale,
			outX = p.outX * scale,
			outY = p.outY * scale,
		})
	end
	return out
end

local KART_CAMERA = CFrame.lookAt(Vector3.new(6.6, 3.4, 7.6), Vector3.new(0, 0.75, 0))
local PART_CAMERA = CFrame.lookAt(Vector3.new(4.4, 3.0, 5.2), Vector3.new(0, 1.0, 0))
local NEON = Enum.Material.Neon
local SMOOTH = Enum.Material.SmoothPlastic
local METAL = Enum.Material.Metal
local BLOCK = Enum.PartType.Block
local CYLINDER = Enum.PartType.Cylinder
local BALL = Enum.PartType.Ball
local FLAT = CFrame.Angles(0, 0, math.rad(90))
local SIDEWAYS = CFrame.Angles(0, math.rad(90), 0)

local function sceneKit(runtime: Facet.Runtime, world: WorldModel): SceneKit
	local H = Facet.host(runtime)
	local parent = C.static(world)
	local spinners: { Spinner } = {}

	local function add(
		name: string,
		shape: Enum.PartType,
		size: Vector3,
		cf: CFrame,
		color: Color3,
		material: Enum.Material,
		transparency: number,
		spins: boolean
	)
		local part = H.Part(name) {
			Parent = parent,
			Anchored = true,
			CanCollide = false,
			CanQuery = false,
			CanTouch = false,
			CastShadow = false,
			Shape = shape,
			Size = size,
			CFrame = cf,
			Color = color,
			Material = material,
			Transparency = transparency,
			TopSurface = Enum.SurfaceType.Smooth,
			BottomSurface = Enum.SurfaceType.Smooth,
		}
		if spins then
			table.insert(spinners, { part = part, base = cf })
		end
	end

	local function wedge(name: string, size: Vector3, cf: CFrame, color: Color3, spins: boolean)
		local part = H.WedgePart(name) {
			Parent = parent,
			Anchored = true,
			CanCollide = false,
			CanQuery = false,
			CanTouch = false,
			CastShadow = false,
			Size = size,
			CFrame = cf,
			Color = color,
			Material = SMOOTH,
		}
		if spins then
			table.insert(spinners, { part = part, base = cf })
		end
	end

	-- Turntable for inspection. It pauses while the Stage is not live and
	-- stays still under reduced motion; the resting pose stays complete.
	local function spin(live: Facet.Readable<boolean>, still: Facet.Readable<boolean>?): () -> ()
		local angle = 0
		local connection = RunService.Heartbeat:Connect(function(dt: number)
			if not live:peek() then
				return
			end
			if still ~= nil and still:peek() then
				return
			end
			angle = (angle + dt * 0.5) % (2 * math.pi)
			local turn = CFrame.Angles(0, angle, 0)
			for _, item in spinners do
				item.part.CFrame = turn * item.base
			end
		end)
		return function()
			connection:Disconnect()
		end
	end

	return { add = add, wedge = wedge, spin = spin }
end

local function addPad(kit: SceneKit)
	kit.add("Pad", CYLINDER, Vector3.new(0.16, 7.4, 7.4), CFrame.new(0, -0.08, 0) * FLAT, Color3.fromRGB(18, 20, 44), SMOOTH, 0, false)
	kit.add("PadRing", CYLINDER, Vector3.new(0.1, 7.8, 7.8), CFrame.new(0, -0.12, 0) * FLAT, CREDIT_TINT, NEON, 0, false)
end

local function buildKart(kit: SceneKit, kart: Kart)
	local shape = CLASS_SHAPE[kart.class]
	local length, width, height, nose = shape.length, shape.width, shape.height, shape.nose
	local dark = kart.body:Lerp(Color3.new(0, 0, 0), 0.72)
	local tire = Color3.fromRGB(22, 22, 30)
	local deck = 0.7
	local bodyLength = length - nose - 0.3
	local bodyBack = -length / 2 + 0.3
	local bodyX = bodyBack + bodyLength / 2
	local bodyY = deck + height / 2
	local wingY = deck + height + 0.55

	addPad(kit)
	kit.add("Chassis", BLOCK, Vector3.new(length, 0.3, width - 0.3), CFrame.new(0, 0.55, 0), dark, METAL, 0, true)
	kit.add("Body", BLOCK, Vector3.new(bodyLength, height, width * 0.72), CFrame.new(bodyX, bodyY, 0), kart.body, SMOOTH, 0, true)
	kit.wedge(
		"Nose",
		Vector3.new(width * 0.72, height, nose),
		CFrame.new(bodyBack + bodyLength + nose / 2, bodyY, 0) * CFrame.Angles(0, math.rad(-90), 0),
		kart.body,
		true
	)
	kit.add("Canopy", BALL, Vector3.one * 0.95, CFrame.new(bodyX + bodyLength * 0.12, deck + height, 0), kart.glow, SMOOTH, 0.25, true)
	kit.add("Stripe", BLOCK, Vector3.new(bodyLength * 0.92, 0.05, 0.22), CFrame.new(bodyX, deck + height + 0.02, 0), kart.glow, NEON, 0, true)
	kit.add("Wing", BLOCK, Vector3.new(0.4, 0.12, width * 0.96), CFrame.new(-length / 2 + 0.3, wingY, 0), kart.glow, NEON, 0, true)
	for _, side in { -1, 1 } do
		kit.add(
			"WingPost",
			BLOCK,
			Vector3.new(0.12, 0.6, 0.12),
			CFrame.new(-length / 2 + 0.34, wingY - 0.32, side * width * 0.3),
			dark,
			SMOOTH,
			0,
			true
		)
	end
	for _, front in { -1, 1 } do
		for _, side in { -1, 1 } do
			local cf = CFrame.new(front * (length / 2 - 0.8), 0.5, side * (width / 2 - 0.12)) * SIDEWAYS
			kit.add("Tire", CYLINDER, Vector3.new(0.5, 1, 1), cf, tire, SMOOTH, 0, true)
			kit.add("Hub", CYLINDER, Vector3.new(0.54, 0.46, 0.46), cf, kart.glow, NEON, 0, true)
		end
	end
	kit.add("Underglow", BLOCK, Vector3.new(length * 0.82, 0.06, width * 0.7), CFrame.new(0, 0.3, 0), kart.glow, NEON, 0.15, true)
	kit.add("Bumper", BLOCK, Vector3.new(0.14, 0.14, width * 0.8), CFrame.new(length / 2 - 0.1, deck + 0.05, 0), kart.glow, NEON, 0, true)
end

local function buildPart(kit: SceneKit, part: Part)

	addPad(kit)

			kit.add("Coil", CYLINDER, Vector3.new(0.18, 1.4, 1.4), CFrame.new(index * 0.55, 1.05, 0), part.color, NEON, 0, true)

		kit.add("Intake", BLOCK, Vector3.new(0.9, 0.24, 0.8), CFrame.new(0, 1.72, 0), part.color, SMOOTH, 0, true)
	elseif part.category == "boosts" then
		kit.add("Tube", CYLINDER, Vector3.new(2.2, 1.0, 1.0), CFrame.new(0.2, 1.1, 0), metal, METAL, 0, true)
		kit.add("Nozzle", CYLINDER, Vector3.new(0.45, 1.3, 1.3), CFrame.new(-1.0, 1.1, 0), part.color, SMOOTH, 0, true)
		kit.add("Flame", BALL, Vector3.one * 0.95, CFrame.new(-1.45, 1.1, 0), part.color, NEON, 0.2, true)
		kit.add("Fin", BLOCK, Vector3.new(0.8, 0.08, 1.8), CFrame.new(0.5, 1.1, 0), part.color, NEON, 0, true)
		kit.add("Cap", BALL, Vector3.one, CFrame.new(1.3, 1.1, 0), metal, SMOOTH, 0, true)
	elseif part.category == "wheels" then
		local cf = CFrame.new(0, 1.15, 0) * SIDEWAYS
		kit.add("Tire", CYLINDER, Vector3.new(0.9, 2.2, 2.2), cf, Color3.fromRGB(22, 22, 30), SMOOTH, 0, true)

		kit.add("Hub", CYLINDER, Vector3.new(1.0, 0.7, 0.7), cf, part.color, NEON, 0, true)
	else
		kit.add("Base", CYLINDER, Vector3.new(0.3, 1.4, 1.4), CFrame.new(0, 0.15, 0) * FLAT, metal, METAL, 0, true)
		kit.add("Orb", BALL, Vector3.one * 1.1, CFrame.new(0, 1.3, 0), part.color, NEON, 0, true)
		kit.add(
			"Halo",
			CYLINDER,
			Vector3.new(0.06, 2.4, 2.4),
			CFrame.new(0, 1.3, 0) * CFrame.Angles(math.rad(20), 0, 0) * FLAT,
			part.color,
			NEON,
			0.35,
			true
		)
	end
end

local function kartStage(kart: Kart, detail: boolean, still: Facet.Readable<boolean>?): ViewportFrame
	return UI.Stage(`{kart.id}Preview`) {
		Size = UDim2.fromScale(1, 1),
		ZIndex = if detail then 2 else 1,
		BackgroundColor3 = Color3.fromRGB(16, 12, 40),
		BackgroundTransparency = if detail then 1 else 0,
		Ambient = Color3.fromRGB(120, 125, 175),
		LightColor = Color3.fromRGB(255, 248, 235),
		LightDirection = Vector3.new(-0.5, -1, -0.7),
		camera = KART_CAMERA,
		fieldOfView = if detail then 32 else 30,
		lazy = not detail,
		content = function(runtime: Facet.Runtime, world: WorldModel, live: Facet.Readable<boolean>): (() -> ())?
			local kit = sceneKit(runtime, world)

local function hub(actions: HubActions): Frame
	local env = UI.environment()
	local barWidth = env.scaled(200)
	local shortBar = env.scaled(90)
	local iconSize = env.scaled(22)
	local columnGap = env.scaled(30)
	local touchOnly = UserInputService.TouchEnabled
		and not UserInputService.MouseEnabled
		and not UserInputService.KeyboardEnabled
	-- Lift the HUD above Roblox's thumbstick and Jump button on touch-only devices.
	local hudPadding: Facet.Sides = {
		top = "m",
		left = "m",
		right = "m",
		bottom = if touchOnly then env.thumbTarget:peek() + 28 else "m",
	}

	local function levelText(use: Facet.Use): string
		return `LV {use(state).level}`
	end
	local function xpText(use: Facet.Use): string
		local snapshot = use(state)
		return `{format.grouped(snapshot.xp)} / {format.grouped(snapshot.xpGoal)}`
	end
	local function xpValue(use: Facet.Use): number
		return use(state).xp
	end
	local function xpMax(use: Facet.Use): number
		return use(state).xpGoal
	end
	local function racesText(use: Facet.Use): string
		local count = use(state).races
		return if count == 1 then "1 race finished" else `{format.grouped(count)} races finished`
	end
	local function barSize(use: Facet.Use): UDim2
		return UDim2.fromOffset(use(barWidth), 0)
	end
	local function shortBarSize(use: Facet.Use): UDim2
		return UDim2.fromOffset(use(shortBar), 0)
	end

	local function raceLabel(use: Facet.Use): string
		local current = use(queue)
		if current.phase == "idle" then
			return "Race"
		elseif current.phase == "joining" then
			return "Joining…"
		elseif current.phase == "queued" then
			return `Leave queue  {format.duration(use(clock) - current.since)}`
		elseif current.phase == "leaving" then
			return "Leaving…"
		elseif current.phase == "found" then
			return "Race found!"
		end
		return "Racing…"
	end
	local function raceBusy(use: Facet.Use): boolean
		local phase = use(queue).phase
		return phase == "joining" or phase == "leaving"
	end
	local function raceEnabled(use: Facet.Use): boolean
		local phase = use(queue).phase
		return phase ~= "found" and phase ~= "racing"
	end
	local function raceAppearance(use: Facet.Use): string
		return if use(queue).phase == "queued" then "standard" else "emphasis"
	end
	local function raceStatus(use: Facet.Use): string
		local current = use(queue)
		local kart = kartById(use(state).kart)
		if current.phase == "joining" or current.phase == "queued" or current.phase == "leaving" then
			return `Searching for racers · {kart.name}`
		elseif current.phase == "found" then
			return "Race found · Neon Circuit · loading"
		elseif current.phase == "racing" then
			return `Racing Neon Circuit in {kart.name}`
		end
		local last = use(lastRace)
		if last ~= "" then
			return last
		end
		return `Next race kart: {kart.name}`
	end
	local function raceButton(name: string): TextButton
		return UI.Button(name) {
			label = raceLabel,
			icon = "flag",
			appearance = raceAppearance,
			controlSize = "large",
			target = "thumb",
			enabled = raceEnabled,
			busy = raceBusy,
			shortcutHint = actions.race,
			onActivate = function()
				pressRace()
			end,
		}
	end

	return Host.Frame "Hub" {
		BackgroundTransparency = 1,
		Size = UDim2.fromScale(1, 1),
		UI.Composition "HUD" {
			padding = hudPadding,
			UI.Region "Race" {
				zone = "bottom",
				rank = 1,
				reveal = false,
				UI.VStack "RaceFull" {
					gap = "s",
					align = "center",
					UI.HStack "RaceStatus" {
						surface = "panel",
						padding = "s",
						UI.Text "Status" { text = raceStatus, textRole = "label" },
					},
					raceButton("RaceCommand"),
				},
				UI.VStack "RaceShort" { align = "center", raceButton("RaceCommandShort") },
			},
			UI.Region "Actions" {
				zone = "right",
				rank = 2,
				reveal = false,
				UI.VStack "ActionColumn" {
					gap = columnGap,
					align = "center",
					padding = { bottom = "l" },
					SelectionGroup = true,
					commandButton(actions.garage, "regular"),
					commandButton(actions.shop, "regular"),
					commandButton(actions.daily, "regular"),
					commandButton(actions.chat, "regular"),
				},
				UI.VStack "ActionGrid" {
					gap = columnGap,
					padding = { bottom = "l" },
					SelectionGroup = true,
					UI.HStack "Upper" {
						gap = "m",
						commandButton(actions.garage, "compact"),
						commandButton(actions.shop, "compact"),
					},
					UI.HStack "Lower" {
						gap = "m",
						commandButton(actions.daily, "compact"),
						commandButton(actions.chat, "compact"),
					},
				},
			},
			UI.Region "Wallet" {
				zone = "topRight",
				rank = 3,
				UI.HStack "WalletFull" {
					surface = "panel",
					padding = "s",
					gap = "m",
					align = "center",
					walletReadout("credits", true, true, iconSize),
					walletReadout("gems", true, true, iconSize),
				},
				UI.HStack "WalletShort" {
					surface = "panel",
					padding = "s",
					gap = "s",
					align = "center",
					walletReadout("credits", true, false, iconSize),
					walletReadout("gems", true, false, iconSize),
				},
			},
			UI.Region "Profile" {
				zone = "topLeft",
				rank = 4,
				UI.VStack "ProfileFull" {
					surface = "panel",
					padding = "m",
					gap = "xs",
					UI.HStack "Identity" {
						gap = "s",
						align = "center",
						UI.Badge "Level" { label = levelText, status = "accent" },
						UI.Text "Name" { text = player.DisplayName, textRole = "strong", AutoLocalize = false },
					},
					UI.ProgressView "XP" { label = "XP", endLabel = xpText, value = xpValue, max = xpMax, Size = barSize },
					UI.Text "Races" { text = racesText, textRole = "caption", role = "secondary" },
				},
				UI.HStack "ProfileShort" {
					surface = "panel",
					padding = "s",
					gap = "s",
					align = "center",
					UI.Badge "Level" { label = levelText, status = "accent" },

local function partCard(part: Part): GuiObject
	return UI.Card(part.id) {
		title = part.name,
		caption = priceText(part),
		imageAspectRatio = 16 / 10,
		artwork = function(): GuiObject
			return partStage(part, false, nil)
		end,
		badges = {
			leading = { label = RARITY_LABEL[part.rarity], status = RARITY_STATUS[part.rarity] },
			trailing = {
				label = function(use: Facet.Use): string?
					local snapshot = use(state)
					if snapshot.owned[part.id] then
						return "Owned"
					end
					if use(purchasePending)[part.id] then
						return "Buying…"
					end
					local missing = part.price - balanceOf(snapshot, part.currency)
					if missing > 0 then
						return `Need {format.grouped(missing)}`
					end
					return nil
				end,
				status = function(use: Facet.Use): Facet.Status
					if use(state).owned[part.id] then
						return "success"
					end
					if use(purchasePending)[part.id] then
						return "info"
					end
					return "warning"
				end,
			},
		},
	}
end

local function partDetail(chosen: Facet.Readable<Part>): Frame
	local env = UI.environment()
	local branches: { [string]: () -> Instance } = {}
	for _, part in PARTS do
		branches[part.id] = function(): Instance
			return partStage(part, true, env.reducedMotion)
		end
	end
	local function chosenId(use: Facet.Use): string
		return use(chosen).id
	end
	local function isOwned(use: Facet.Use): boolean
		return use(state).owned[use(chosen).id] == true
	end
	local function isPending(use: Facet.Use): boolean
		return use(purchasePending)[use(chosen).id] == true
	end
	local function buyLabel(use: Facet.Use): string
		if isOwned(use) then
			return "Owned"
		end
		if isPending(use) then
			return "Buying…"
		end
		return "Buy"
	end
	local function balanceText(use: Facet.Use): string
		local part = use(chosen)
		local snapshot = use(state)
		if snapshot.owned[part.id] then
			return "Already in your parts bay"
		end
		if use(purchasePending)[part.id] then
			return "Waiting for the shop to confirm…"
		end
		local balance = balanceOf(snapshot, part.currency)
		local unit = currencyName(part.currency, 2)
		if balance >= part.price then
			return `You have {format.grouped(balance)} {unit} · {format.grouped(balance - part.price)} left after`
		end
		return `You have {format.grouped(balance)} {unit} · need {format.grouped(part.price - balance)} more`
	end

	return UI.VStack "PartDetail" {
		width = "fill",
		height = "fill",
		gap = "s",
		padding = "m",
		UI.HStack "Identity" {
			width = "fill",
			gap = "s",
			align = "center",
			UI.VStack "Names" {
				gap = "xs",
				UI.fill(),
				UI.Text "Name" {
					text = function(use: Facet.Use): string
						return use(chosen).name
					end,
					textRole = "heading",
					wrap = true,
				},
				UI.Text "Kind" {
					text = function(use: Facet.Use): string
						local part = use(chosen)
						return `{CATEGORY_LABEL[part.category]} · {RARITY_LABEL[part.rarity]}`
					end,
					textRole = "caption",
					role = "secondary",
				},
			},
			UI.Badge "Effect" {
				label = function(use: Facet.Use): string?
					return use(chosen).effect
				end,
				status = "accent",
			},
		},
		UI.ZStack "Art" { width = "fill", height = "fill", UI.fill(), showroomBackdrop(), C.switch(chosenId, branches) },
		UI.Text "About" {
			text = function(use: Facet.Use): string
				return use(chosen).blurb
			end,
			wrap = true,
		},
		UI.HStack "Commit" {
			width = "fill",
			gap = "m",
			align = "center",
			UI.VStack "Terms" {
				gap = "xs",
				UI.fill(),
				UI.Text "Price" {
					text = function(use: Facet.Use): string
						return priceText(use(chosen))
					end,
					textRole = "numeral",
				},
				UI.Text "Balance" { text = balanceText, textRole = "caption", role = "secondary", wrap = true },
			},
			UI.Button "Buy" {
				label = buyLabel,
				appearance = "emphasis",
				controlSize = "large",
				enabled = function(use: Facet.Use): boolean
					return not isOwned(use)
				end,
				busy = isPending,
				onActivate = function()
					requestBuy(chosen:peek())
				end,
			},
		},
		C.show(function(use: Facet.Use): boolean
			local refusal = use(shopRefusal)
			return refusal.message ~= "" and refusal.id == use(chosen).id
		end, function()
			return UI.Notice "Refusal" {
				message = function(use: Facet.Use): string
					return use(shopRefusal).message
				end,
				severity = "error",
			}
		end),
	}
end

local function shopBody(back: Facet.Cell<boolean>): Frame
	local env = UI.environment()
	local visible = C.formula(function(use: Facet.Use): { Part }
		local category = use(shopCategory)
		local list: { Part } = {}
		for _, part in PARTS do
			if category == "all" or part.category == category then
				table.insert(list, part)
			end
		end
		return list
	end)
	local chosen = C.formula(function(use: Facet.Use): Part
		for id, on in use(shopSelection) do
			local part = PART_BY_ID[id]
			if on and part then
				return part
			end
		end
		return PARTS[1]
	end)
	local categorySpec: Facet.SinglePickerSpec<string> =
		{ selected = shopCategory, options = CATEGORY_OPTIONS, style = "segmented" }

	return UI.NavigationSplitView "ShopSplit" {
		preferredCompactColumn = shopColumn,
		resizable = false,
		detailLabel = false,
		compactBack = back,
		sidebar = function(): Instance
			local spec: Facet.KeyedCollectionSpec<Part, string> = {
				UI.fill(),
				from = visible,
				key = "id",
				selection = shopSelection,
				selectionMode = "single",
				minColumnWidth = env.scaled(168),
				itemSize = env.scaled(200),
				measure = true,
				gap = env.scaled(12),
				onActivate = function()
					shopColumn:set("detail")
				end,
				render = function(row: Facet.Readable<Part>): GuiObject
					return partCard(row:peek())
				end,
			}
			return UI.VStack "Browse" {
				width = "fill",
				height = "fill",
				gap = "s",
				padding = "s",
				align = "stretch",
				UI.Picker "Category" (categorySpec),
				UI.Text "Count" {
					text = function(use: Facet.Use): string
						local count = #use(visible)
						return if count == 1 then "1 part" else `{count} parts`
					end,
					textRole = "caption",
					role = "secondary",
				},
				UI.VirtualGrid "Parts" (spec),
			}
		end,
		detail = function(): Instance
			return partDetail(chosen)
		end,
	}
end

local function kartCard(kart: Kart): GuiObject
	return UI.Card(kart.id) {
		title = kart.name,
		caption = `{CLASS_LABEL[kart.class]} · Speed {kart.speed}`,
		imageAspectRatio = 16 / 10,
		artwork = function(): GuiObject
			return kartStage(kart, false, nil)
		end,
		badges = {
			leading = { label = RARITY_LABEL[kart.rarity], status = RARITY_STATUS[kart.rarity] },
			trailing = {
				label = function(use: Facet.Use): string?
					if use(state).kart == kart.id then
						return "Next race"
					end
					if use(favorites)[kart.id] then
						return "Favorite"
					end
					return nil
				end,
				status = function(use: Facet.Use): Facet.Status
					return if use(state).kart == kart.id then "success" else "warning"
				end,
			},
		},
	}
end

local function emptyState(): Frame
	return UI.VStack "NoMatches" {
		width = "fill",
		surface = "panel",
		padding = "m",
		gap = "s",
		UI.Text "Title" { text = "No karts match", textRole = "heading" },
		UI.Text "Why" {
			text = function(use: Facet.Use): string
				local filters: { string } = {}
				local kind = use(garageClass)
				if kind ~= "all" then
					table.insert(filters, CLASS_PLURAL[kind] or kind)
				end
				local rarity = use(garageRarity)
				if rarity ~= "all" then
					table.insert(filters, RARITY_LABEL[rarity] or rarity)
				end
				local scope = if #filters > 0 then table.concat(filters, " · ") else "any class or rarity"
				local query = trim(use(garageQuery))
				if query ~= "" then
					return `No kart name contains “{query}” in {scope}.`
				end
				return `There are no karts in {scope}.`
			end,
			role = "secondary",
			wrap = true,
		},
		UI.HStack "Fixes" {
			gap = "s",
			wrap = true,
			UI.Button "ClearSearch" {
				label = "Clear search",
				enabled = function(use: Facet.Use): boolean
					return use(garageQuery) ~= ""
				end,
				onActivate = function()
					garageQuery:set("")
				end,
			},
			UI.Button "ResetFilters" {
				label = "Reset filters",
				enabled = function(use: Facet.Use): boolean
					return use(garageClass) ~= "all" or use(garageRarity) ~= "all"
				end,
				onActivate = function()
					garageClass:set("all")
					garageRarity:set("all")
				end,
			},
		},
	}
end

local function kartDetail(chosen: Facet.Readable<Kart>): Frame
	local env = UI.environment()
	local equipped = C.formula(function(use: Facet.Use): Kart
		return kartById(use(state).kart)
	end)
	local branches: { [string]: () -> Instance } = {}
	for _, kart in KARTS do
		branches[kart.id] = function(): Instance
			return kartStage(kart, true, env.reducedMotion)
		end
	end
	local function chosenId(use: Facet.Use): string
		return use(chosen).id
	end
	local function isEquipped(use: Facet.Use): boolean
		return use(state).kart == use(chosen).id
	end
	local function isFavorite(use: Facet.Use): boolean
		return use(favorites)[use(chosen).id] == true
	end
	local function setFavorite(on: boolean)
		local id = chosen:peek().id
		local marks = table.clone(favorites:peek())
		if on then
			marks[id] = true
		else
			marks[id] = nil
		end
		favorites:set(marks)
	end
	local function stat(name: string, pick: (Kart) -> number): Frame
		return UI.ProgressView(name) {
			label = name,
			max = 100,
			value = function(use: Facet.Use): number
				return pick(use(chosen))
			end,
			endLabel = function(use: Facet.Use): string
				local kart, current = use(chosen), use(equipped)
				local value = pick(kart)
				local delta = value - pick(current)
				if kart.id == current.id or delta == 0 then
					return tostring(value)
				end
				local sign = if delta > 0 then "+" else ""
				return `{value} ({sign}{delta})`
			end,
			Size = UDim2.new(1, 0, 0, 0),
		}
	end
	local function chooseLabel(use: Facet.Use): string
		local kart = use(chosen)
		if use(state).kart == kart.id then
			return "Set for your next race"
		end
		if use(kartPending) == kart.id then
			return "Locking in…"
		end
		return `Race with {kart.name}`
	end

	local wide = C.cell(false)
	local function basis(use: Facet.Use): UDim2
		return if use(wide) then UDim2.new(0, 0, 1, 0) else UDim2.new(1, 0, 0, 0)
	end
	local facts = UI.VStack "Facts" {
		width = "fill",
		gap = "s",
		align = "stretch",
		stat("Speed", function(kart: Kart): number
			return kart.speed
		end),
		stat("Handling", function(kart: Kart): number
			return kart.handling
		end),
		stat("Boost", function(kart: Kart): number
			return kart.boost
		end),
		UI.Text "Compare" {
			text = function(use: Facet.Use): string
				local kart, current = use(chosen), use(equipped)
				if kart.id == current.id then
					return "This is your next-race kart."
				end
				return `Numbers in brackets compare with your next-race kart, {current.name}.`
			end,
			textRole = "caption",
			role = "secondary",
			wrap = true,
		},
		UI.Text "About" {
			text = function(use: Facet.Use): string
				return use(chosen).blurb
			end,
			wrap = true,
		},
		UI.Toggle "Favorite" { label = "Favorite", presentation = "checkbox", value = isFavorite, onChange = setFavorite },
	}
	local panes = UI.AdaptiveStack "Panes" {
		width = "fill",
		height = "fill",
		gap = "m",
		UI.fill(),
		axis = function(use: Facet.Use): "x" | "y"
			return if use(wide) then "x" else "y"
		end,
		UI.ZStack "Art" {
			Size = basis,
			AutomaticSize = Enum.AutomaticSize.None,
			UI.fill(3),
			showroomBackdrop(),
			C.switch(chosenId, branches),
		},
		UI.ScrollView "Fields" {
			Size = basis,
			AutomaticSize = Enum.AutomaticSize.None,
			UI.fill(2),
			padding = { right = "s" },
			facts,
		},
	}
	local room = UI.environment(panes)
	local threshold = room.scaled(620)
	C.watch(function(use: Facet.Use)
		wide:set(use(room.viewportWidth) >= use(threshold))
	end)

	return UI.VStack "KartDetail" {
		width = "fill",
		height = "fill",
		gap = "s",
		padding = "m",
		align = "stretch",
		UI.HStack "Identity" {
			width = "fill",
			gap = "s",
			align = "center",
			UI.VStack "Names" {
				gap = "xs",
				UI.fill(),
				UI.Text "Name" {
					text = function(use: Facet.Use): string
						return use(chosen).name
					end,
					textRole = "heading",
					wrap = true,
				},
				UI.Text "Class" {
					text = function(use: Facet.Use): string
						local kart = use(chosen)
						local notes = CLASS_LABEL[kart.class]
						if isFavorite(use) then
							notes ..= " · Favorite"
						end
						if isEquipped(use) then
							notes ..= " · Next race"
						end
						return notes
					end,
					textRole = "caption",
					role = "secondary",
				},
			},
			UI.Badge "Rarity" {
				label = function(use: Facet.Use): string?
					return RARITY_LABEL[use(chosen).rarity]
				end,
				status = function(use: Facet.Use): Facet.Status
					return RARITY_STATUS[use(chosen).rarity]
				end,
			},
		},
		panes,
		UI.Button "Choose" {
			label = chooseLabel,
			appearance = function(use: Facet.Use): string
				return if isEquipped(use) then "standard" else "emphasis"
			end,
			controlSize = "large",
			enabled = function(use: Facet.Use): boolean
				return not isEquipped(use)
			end,
			busy = function(use: Facet.Use): boolean
				return use(kartPending) ~= nil
			end,
			onActivate = function()
				chooseKart(chosen:peek())
			end,
		},
		C.show(function(use: Facet.Use): boolean
			return use(kartRefusal) ~= ""
		end, function()
			return UI.Notice "Refusal" { message = kartRefusal, severity = "error" }
		end),
	}
end

local function garageBody(back: Facet.Cell<boolean>): Frame
	local env = UI.environment()
	local visible = C.formula(function(use: Facet.Use): { Kart }
		return filterKarts(use(garageQuery), use(garageClass), use(garageRarity), use(garageSort))
	end)
	local chosen = C.formula(function(use: Facet.Use): Kart
		for id, on in use(garageSelection) do
			local kart = KART_BY_ID[id]
			if on and kart then
				return kart
			end
		end
		return kartById(DEFAULT_KART)
	end)
	local classSpec: Facet.SinglePickerSpec<string> =
		{ selected = garageClass, options = CLASS_OPTIONS, style = "menu", sizing = "hug" }
	local raritySpec: Facet.SinglePickerSpec<string> =
		{ selected = garageRarity, options = RARITY_OPTIONS, style = "menu", sizing = "hug" }
	local sortSpec: Facet.SinglePickerSpec<string> =
		{ selected = garageSort, options = SORT_OPTIONS, style = "menu", sizing = "hug" }
	local function countText(use: Facet.Use): string
		local count = #use(visible)
		if count == #KARTS then
			return `{count} karts`
		end
		return `{count} of {#KARTS} karts`
	end
	local function grid(): ScrollingFrame
		local spec: Facet.KeyedCollectionSpec<Kart, string> = {
			UI.fill(),
			from = visible,
			key = "id",
			selection = garageSelection,
			selectionMode = "single",
			minColumnWidth = env.scaled(168),
			itemSize = env.scaled(200),
			measure = true,
			gap = env.scaled(12),
			onActivate = function()
				garageColumn:set("detail")
			end,
			render = function(row: Facet.Readable<Kart>): GuiObject
				return kartCard(row:peek())
			end,
		}
		return UI.VirtualGrid "Karts" (spec)
	end

	return UI.NavigationSplitView "GarageSplit" {
		preferredCompactColumn = garageColumn,
		resizable = false,
		detailLabel = false,
		compactBack = back,
		sidebar = function(): Instance
			return UI.VStack "Browse" {
				width = "fill",
				height = "fill",
				gap = "s",
				padding = "s",
				align = "stretch",
				UI.TextInput "Search" { value = garageQuery, placeholder = "Search karts by name", presentation = "search" },
				UI.HStack "Filters" {
					gap = "s",
					wrap = true,
					UI.Picker "Class" (classSpec),
					UI.Picker "Rarity" (raritySpec),
					UI.Picker "Sort" (sortSpec),
				},
				UI.Text "Count" { text = countText, textRole = "caption", role = "secondary" },
				C.show(function(use: Facet.Use): boolean
					return #use(visible) > 0
				end, grid, emptyState),
			}
		end,
		detail = function(): Instance
			return kartDetail(chosen)
		end,
	}
end

local function garagePage(): Frame
	local back = C.cell(false)
	return UI.Screen "GaragePage" {
		surface = "surface",
		UI.NavBar "GarageBar" { title = "Garage", onBack = back, trailing = closeButton() },
		UI.VStack "Body" { width = "fill", height = "fill", UI.fill(), garageBody(back) },
	}
end

-- Mount: one app, one tree

local function gameplayGuard(run: () -> ()): () -> ()
	return function()
		if UserInputService:GetFocusedTextBox() ~= nil then
			return
		end
		run()
	end
end

app.mount(function()
	local hasPage = C.formula(function(use: Facet.Use): boolean
		return #use(path) > 0
	end)
	C.watch(function(use: Facet.Use)
		if use(hasPage) then
			C.cleanup(suspendGameplay())
		end
	end)

	local ticking = true
	C.cleanup(function()
		ticking = false
	end)
	task.spawn(function()
		while ticking and alive do
			task.wait(1)
			if ticking and alive then
				clock:set(os.clock())
			end
		end
	end)

	local function gameAction(name: string, key: Enum.KeyCode, pad: Enum.KeyCode, run: () -> ()): InputAction
		local guarded = gameplayGuard(run)
		return Host.InputAction(name) {
			Type = Enum.InputActionType.Bool,
			Pressed = function()
				guarded()
			end,
			Host.InputBinding "Keyboard" { KeyCode = key },
			Host.InputBinding "Gamepad" { KeyCode = pad },
		}
	end

	local raceAction = gameAction("Race", Enum.KeyCode.R, Enum.KeyCode.ButtonX, pressRace)
	local garageAction = gameAction("Garage", Enum.KeyCode.G, Enum.KeyCode.ButtonY, openGarage)
	local shopAction = gameAction("Shop", Enum.KeyCode.B, Enum.KeyCode.ButtonR1, openShop)
	local dailyAction = gameAction("Daily", Enum.KeyCode.J, Enum.KeyCode.ButtonL1, openDaily)
	local chatAction = gameAction("QuickChat", Enum.KeyCode.T, Enum.KeyCode.ButtonL2, toggleChat)

	-- Hub commands. Disabled while a page or the daily dialog owns input.
	Host.InputContext "NeonCircuitHub" {
		Parent = C.static(playerGui),
		Enabled = function(use: Facet.Use): boolean
			return not use(hasPage) and not use(dailyOpen)
		end,
		raceAction,
		garageAction,
		shopAction,
		dailyAction,
		chatAction,
	}

	local actions: HubActions = {
		race = raceAction,
		garage = {
			id = "Garage",
			label = "Garage",
			caption = "Garage",
			glyph = garageGlyph,
			action = garageAction,
			run = openGarage,
		},
		shop = {
			id = "Shop",
			label = "Parts shop",
			caption = "Shop",
			glyph = shopGlyph,
			action = shopAction,
			run = openShop,
		},
		daily = {
			id = "Daily",
			label = "Daily rewards",
			caption = function(use: Facet.Use): string
				return if use(state).dailyReady then "Ready!" else "Daily"
			end,
			glyph = dailyGlyph,
			action = dailyAction,
			run = openDaily,
		},
		chat = {
			id = "Chat",
			label = "Quick chat",
			caption = "Chat",
			glyph = chatGlyph,
			action = chatAction,
			run = toggleChat,
		},
	}

	return UI.NavigationStack "Navigation" {
		path = path,
		chrome = "hidden",
		root = {
			content = function(): Instance
				return hub(actions)
			end,
		},
		destinations = {
			shop = {
				initialFocus = "Body",
				content = function(): Instance
					return shopPage()
				end,
			},
			garage = {
				initialFocus = "Body",
				content = function(): Instance
					return garagePage()
				end,
			},
		},
	}
end)

---------------------------------------------------------------------------
-- Evaluation entry points
---------------------------------------------------------------------------

local function showScreen(target: string)
	if not alive then
		return
	end
	resetModel()
	GuiService.SelectedObject = nil
	if target == "hub" then
		path:set({})
	elseif target == "shop" then
		openShop()
	elseif target == "garage" then
		openGarage()
	elseif target == "purchase" then
		openShop()
		local part = PART_BY_ID[SHOWCASE_PART]
		shopSelection:set({ [part.id] = true })
		shopColumn:set("detail")
		requestBuy(part)
	else
		warn(`[NeonCircuit] unknown screen "{target}"; use hub, shop, garage or purchase`)
	end
end

local function disposeAll()
	if not alive then
		return
	end
	alive = false
	epoch += 1
	GuiService.SelectedObject = nil
	app.dispose()
end

if typeof(script) == "Instance" then
	game.Destroying:Connect(disposeAll)
end

_G.EvalUI = {
	show = function(state: string)
		showScreen(state)
	end,
	dispose = function()
		disposeAll()
	end,
}

return nil
