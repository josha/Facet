--!strict

local Play = require("../../../../../../../../../tools/play/host/play")
local QA = require("../../../../../../../../../tools/play/host/qa")

local function tap(qa: QA.QA, query: QA.Query): (boolean, string)
	local pressed, reason = qa:tap(query)
	if pressed then
		qa:sleep(1)
	end
	return pressed, reason
end


local sessions: { Play.Session } = {}
for _, device in { "desktop", "phone-portrait", "phone-landscape", "tablet", "console" } do
	table.insert(sessions, {
		device = device,
		cases = {
			{
				id = "diagnostic/hub",
				run = function(qa, t)
					t:expect(qa:waitText("chase", 10) ~= nil):toBe(true)
					qa:capture("diagnostic-hub")
					qa:dump("diagnostic-hub")
				end,
			},
			{
				id = "diagnostic/shop-cancel",
				run = function(qa, t)
					t:expect(tap(qa, { exact = "Shop" })):toBe(true)
					t:expect(qa:waitText("ion motor", 3) ~= nil):toBe(true)
					qa:capture("diagnostic-shop")
					qa:dump("diagnostic-shop")
					t:expect(tap(qa, { exact = "Purchase", index = 1 })):toBe(true)
					t:expect(qa:waitText("full price: 650", 3) ~= nil):toBe(true)
					t:expect(tap(qa, { exact = "Cancel" })):toBe(true)
					t:expect(qa:hasText("2400 cr") ~= nil):toBe(true)
					t:expect(tap(qa, { exact = "×" })):toBe(true)
				end,
			},
			{
				id = "diagnostic/garage-search-clear",
				run = function(qa, t)
					t:expect(tap(qa, { exact = "Garage" })):toBe(true)
					qa:capture("diagnostic-garage")
					t:expect(qa:typeText({ role = "input" }, "missing-kart", true)):toBe(true)
					t:expect(qa:waitText("no karts found", 3) ~= nil):toBe(true)
					t:expect(tap(qa, { exact = "Clear search" })):toBe(true)
					t:expect(qa:waitGone({ text = "No karts found" }, 3)):toBe(true)
					qa:dump("diagnostic-garage")
					t:expect(tap(qa, { exact = "×" })):toBe(true)
				end,
			},
			{
				id = "diagnostic/livery-draft-cancel",
				run = function(qa, t)
					t:expect(tap(qa, { exact = "Livery" })):toBe(true)
					t:expect(tap(qa, { exact = "Red" })):toBe(true)
					t:expect(qa:waitText("unsaved", 3) ~= nil):toBe(true)
					qa:capture("diagnostic-livery")
					qa:dump("diagnostic-livery")
					t:expect(tap(qa, { exact = "×" })):toBe(true)
					t:expect(qa:waitText("discard unsaved", 3) ~= nil):toBe(true)
					t:expect(tap(qa, { exact = "Cancel" })):toBe(true)
					t:expect(qa:hasText("unsaved") ~= nil):toBe(true)
				end,
			},
		},
	})
end

Play.main({ sessions = sessions }, debug.info(1, "s"))
